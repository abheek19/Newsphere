Connects to your N8N automation workflow that scrapes and transforms news articles into JSON schema { "Headline", "Image URL", "Author", "Article URL", "Category" }.

Executes the custom N8N regex extraction logic in real time: extracts headlines, relative image URLs with data-original, authors with class="person-name...", and article destination links.

Here is the N8N workflow blueprint using your provided regex code node to scrape news, append rows to Google Sheets, and output to this web portal:

\n// HEADLINE\nconst html = $input.first().json.data;\nconst results = [];\nconst seenUrls = new Set();\nconst titleRegex = /

\\s*\\s*([\\s\\S]*?)\\s*<\\/a>\\s*<\\/h3>/g;\nlet match;\nwhile ((match = titleRegex.exec(html)) !== null) {\n const articleUrl = match[1].replace(/&/g, '&');\n const headline = match[2].replace(/\\s+/g, ' ').trim();\n if (!headline || seenUrls.has(articleUrl)) continue;\n seenUrls.add(articleUrl);\n const windowStart = Math.max(0, match.index - 3000);\n const context = html.slice(windowStart, match.index);\n const imgMatches = [...context.matchAll(/data-original=\"([^\"]+)\"/g)];\n const imageUrl = imgMatches.length ? imgMatches[imgMatches.length - 1][1].replace(/&/g, '&') : null;\n const authorMatches = [...context.matchAll(/class=\"person-name[^\"]*\"[^>]*>\\s*([^<]+?)\\s*<\\/a>/g)];\n const author = authorMatches.length ? authorMatches[authorMatches.length - 1][1].trim() : null;\n results.push({ \"Headline\": headline, \"Image URL\": imageUrl, \"Author\": author, \"Article URL\": articleUrl });\n}\nreturn results.map(item => ({ json: item }));" } }, { "name": "Google Sheets - Append Rows", "type": "n8n-nodes-base.googleSheets", "parameters": { "operation": "append", "sheetId": "YOUR_SHEET_ID" } } ] }
